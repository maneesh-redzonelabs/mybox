MyBox Portal — PostgreSQL Database Schema

Relational schema derived from MyBox.html (prototype): users, family members, insurance policies and vehicles.

Overview

The prototype stores one record per mobile number in localStorage, with each user owning three child collections: family members, insurance policies and vehicles. This schema normalizes that structure into four PostgreSQL tables: users (parent) and family_members, policies, vehicles (children), each linked back to users via a foreign key. Fields that the app computes on the fly (Age, Maturity Date, Next Premium Date) are left out of storage and are expected to be derived in application code or SQL views, since they depend on "today's date".

Relationships

  • users.id → family_members.user_id  (one user has many family members)
  • users.id → policies.user_id  (one user has many policies)
  • users.id → vehicles.user_id  (one user has many vehicles)

SQL Schema (PostgreSQL)

-- =========================================================
-- MyBox Portal: PostgreSQL schema
-- =========================================================

CREATE TABLE users (
    id              BIGSERIAL PRIMARY KEY,
    mobile          VARCHAR(10) NOT NULL UNIQUE
                        CHECK (mobile ~ '^[0-9]{10}$'),
    name            VARCHAR(150) NOT NULL,
    email           VARCHAR(150) NOT NULL
                        CHECK (email ~ '^[^@\s]+@[^@\s]+\.[^@\s]+$'),
    date_of_birth   DATE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE family_members (
    id               BIGSERIAL PRIMARY KEY,
    user_id          BIGINT NOT NULL
                         REFERENCES users(id) ON DELETE CASCADE,
    name             VARCHAR(150) NOT NULL,
    date_of_birth    DATE,
    relationship     VARCHAR(30) NOT NULL
                         CHECK (relationship IN (
                             'Self', 'Spouse', 'Son', 'Daughter', 'Father',
                             'Mother', 'Brother', 'Sister', 'Doughter-in-law',
                             'Son-in-law', 'Uncle', 'Aunt',
                             'Cousin Sister', 'Cousin Brother'
                         )),
    mobile           VARCHAR(10)
                         CHECK (mobile IS NULL OR mobile ~ '^[0-9]{10}$'),
    living_together  VARCHAR(3) NOT NULL
                         CHECK (living_together IN ('Yes', 'No')),
    created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_family_members_user_id ON family_members(user_id);

CREATE TABLE policies (
    id                  BIGSERIAL PRIMARY KEY,
    user_id             BIGINT NOT NULL
                            REFERENCES users(id) ON DELETE CASCADE,
    policy_no           VARCHAR(50) NOT NULL,
    bank_name           VARCHAR(150) NOT NULL,
    policy_type         VARCHAR(20) NOT NULL
                            CHECK (policy_type IN ('Life Insurance', 'Health Insurance')),
    start_date          DATE NOT NULL,
    policy_term_years   SMALLINT NOT NULL CHECK (policy_term_years BETWEEN 1 AND 60),
    paying_term_years   SMALLINT NOT NULL CHECK (paying_term_years BETWEEN 1 AND 60),
    emi_frequency       VARCHAR(20) NOT NULL
                            CHECK (emi_frequency IN (
                                'Monthly', 'Quaterly', 'Half Yearly',
                                'Yearly', 'Single Premium'
                            )),
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (user_id, policy_no)
);

CREATE INDEX idx_policies_user_id ON policies(user_id);

CREATE TABLE vehicles (
    id                 BIGSERIAL PRIMARY KEY,
    user_id            BIGINT NOT NULL
                           REFERENCES users(id) ON DELETE CASCADE,
    registration_no    VARCHAR(15) NOT NULL UNIQUE
                           CHECK (registration_no ~ '^[A-Z]{2}[0-9]{1,2}[A-Z]{1,3}[0-9]{4}$'),
    vehicle_type       VARCHAR(10) NOT NULL
                           CHECK (vehicle_type IN ('Car', 'Bike', 'Scooty')),
    make               VARCHAR(100) NOT NULL,
    model_year         SMALLINT NOT NULL
                           CHECK (model_year BETWEEN 1990 AND 2100),
    insurance_policy   VARCHAR(100),
    pollution_upto     DATE,
    created_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at         TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_vehicles_user_id ON vehicles(user_id);

-- =========================================================
-- Keep updated_at current on every UPDATE
-- =========================================================
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_family_members_updated_at
    BEFORE UPDATE ON family_members
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_policies_updated_at
    BEFORE UPDATE ON policies
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_vehicles_updated_at
    BEFORE UPDATE ON vehicles
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- =========================================================
-- Optional: views for the app's computed/display fields
-- =========================================================
CREATE OR REPLACE VIEW v_family_members AS
SELECT fm.*,
       DATE_PART('year', AGE(fm.date_of_birth)) AS age
FROM family_members fm;

CREATE OR REPLACE VIEW v_policies AS
SELECT p.*,
       (p.start_date + (p.policy_term_years || ' years')::interval)::date AS maturity_date
FROM policies p;
Note on policy_no: marked unique per-user (UNIQUE (user_id, policy_no)) rather than globally unique, since two different customers of the same bank could in theory be issued overlapping identifiers in mock/test data. Switch to a plain UNIQUE column if policy numbers are guaranteed globally unique in production. Next Premium Date is intentionally not materialized as a view column because it depends on "today's date" at read time — compute it in application code or as a scalar function called per-request.
